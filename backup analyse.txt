    Image as ImageIcon,

    ChevronUp,

    Clipboard,

    Info

/* ── Types ── */
interface ECGAnalysisData {
    _id: string;
    ecg: {
        _id: string;
        title: string;
        originalImage: string;
        patient?: { fullName?: string };
        createdAt?: string;
    };
    plot12leads?: string;
    plotFullLeadII?: string;
    plotImage?: string;
    npyFile?: string;
    aiResult?: any;
    doctorNotes?: string;
    status: 'uploaded' | 'digitized' | 'analyzed';
    createdAt?: string;
}

const CATEGORY_LABELS: Record<string, string> = {
    cd: 'Troubles de conduction',
    arrhythmia: 'Arythmies',
    ihd: 'Ischémie cardiaque',
    beat: 'Battements anormaux',
    hyp: 'Hypertrophie',
    morphology: 'Morphologie',
};

/* ── Components ── */

const PremiumCard = ({ children, className = "" }: { children: React.ReactNode, className?: string }) => (
  <div className={`bg-white/70 backdrop-blur-xl border border-white/50 rounded-[36px] shadow-[0_15px_40px_rgba(0,0,0,0.04)] overflow-hidden ${className}`}>
    {children}
  </div>
);

const MetricBox = ({ label, value, unit, status, icon: Icon }: any) => {
    const getColors = () => {
        if (status === 'anomaly') return 'text-red-600 bg-red-50/50 border-red-100';
        if (status === 'warning') return 'text-amber-600 bg-amber-50/50 border-amber-100';
        return 'text-indigo-600 bg-indigo-50/50 border-indigo-100';
    };
    
    return (
        <div className={`p-5 rounded-3xl border ${getColors()} flex flex-col gap-1 transition-all hover:scale-[1.02] cursor-default`}>
            <div className="flex items-center justify-between mb-1">
                <span className="text-[10px] font-black uppercase tracking-widest opacity-60">{label}</span>
                {Icon && <Icon className="w-3.5 h-3.5 opacity-40" />}
            </div>
            <div className="flex items-baseline gap-1">
                <span className="text-2xl font-black tracking-tight">
                    {typeof value === 'number' ? value.toFixed(2) : value || '—'}
                </span>
                {unit && <span className="text-[10px] font-bold uppercase opacity-60">{unit}</span>}
            </div>
        </div>
    );
};

/* ── Helpers ── */
function extractMetrics(aiResult: any) {
    if (!aiResult) return null;
    const flat = typeof aiResult === 'object' ? aiResult : {};
    return {
        hr: flat.heart_rate ?? flat.hr ?? flat.HR ?? null,
        pr: flat.pr_interval ?? flat.pr ?? flat.PR ?? null,
        qrs: flat.qrs_duration ?? flat.qrs ?? flat.QRS ?? null,
        qtc: flat.qtc ?? flat.QTc ?? flat.QTC ?? null,
        rhythm: flat.rhythm ?? flat.Rhythm ?? null,
        diagnosis: flat.diagnosis ?? flat.prediction ?? flat.label ?? null,
        deterministic: flat.deterministic ?? flat.rule_based ?? null,
        ai_classification: flat.ai_classification ?? null,
    };
}

export default function ECGAnalysis() {
    const { id } = useParams<{ id: string }>();
    const navigate = useNavigate();
    const location = useLocation();

    const [analysis, setAnalysis] = useState<ECGAnalysisData | null>(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);

    const [digitizing, setDigitizing] = useState(false);
    const [analyzing, setAnalyzing] = useState(false);
    const [notes, setNotes] = useState('');
    const [savingNotes, setSavingNotes] = useState(false);
    
    const [activePlot, setActivePlot] = useState(0);

    /* ── Fetch ── */
    const fetchAnalysis = async () => {
        if (!id) return;
        try {
            setLoading(true);
            const data = await getECGAnalysisDetails(id);
            setAnalysis(data);
            setNotes(data.doctorNotes || '');
        } catch (err: any) {
            setError(err.message || 'Erreur chargement');
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        fetchAnalysis();
    }, [id]);

    /* ── Actions ── */
    const handleDigitize = async () => {
        if (!analysis?._id) return;
        try {
            setDigitizing(true);
            await digitizeECGAnalysis(analysis._id);
            await fetchAnalysis();
        } catch (err: any) {
            alert(err.message);
        } finally {
            setDigitizing(false);
        }
    };

    const handleAnalyze = async () => {
        if (!analysis?._id) return;
        try {
            setAnalyzing(true);
            await analyzeECGWithAI(analysis._id);
            await fetchAnalysis();
        } catch (err: any) {
            alert(err.message);
        } finally {
            setAnalyzing(false);
        }
    };

    const handleSaveNotes = async () => {
        if (!id) return;
        try {
            setSavingNotes(true);
            await saveDoctorNotes(id, notes);
        } catch (err) {
            console.error(err);
        } finally {
            setSavingNotes(false);
        }
    };

    const isDigitized = analysis?.status === 'digitized' || analysis?.status === 'analyzed';
    const isAnalyzed = analysis?.status === 'analyzed';
    const metrics = useMemo(() => extractMetrics(analysis?.aiResult), [analysis]);
    
    const imageUrl = useMemo(() => {
        const rawUrl = analysis?.ecg?.originalImage || '';
        return location.state?.imageUrl ? getImageUrl(location.state.imageUrl) : getImageUrl(rawUrl);
    }, [analysis, location.state]);

    const plots = useMemo(() => {
      if (!analysis) return [];
      const list = [];
      if (analysis.plot12leads) list.push({ url: analysis.plot12leads, label: "12 Dérivations" });
      if (analysis.plotFullLeadII) list.push({ url: analysis.plotFullLeadII, label: "Lead II Long" });
      if (analysis.plotImage) list.push({ url: analysis.plotImage, label: "Résumé 4 Pistes" });
      return list;
    }, [analysis]);

    if (loading) {
        return (
            <DashboardLayout>
                <div className="flex items-center justify-center py-40">
                    <Loader2 className="w-10 h-10 animate-spin text-indigo-500" />
                </div>
            </DashboardLayout>
        );
    }

    if (error || !analysis) {
        return (
            <DashboardLayout>
                <div className="p-8 text-center text-red-500 font-bold">
                    {error || "Analyse introuvable"}
                </div>
            </DashboardLayout>
        );
    }

    const aiStatus = metrics?.ai_classification?.status || 'NORMAL';
    const isAnomalous = aiStatus === 'ANORMAL';

    return (
        <DashboardLayout>
            <div className="p-8 space-y-8">
                
                {/* ── HERO HEADER ── */}
                <div className="relative overflow-hidden rounded-[40px] bg-gradient-to-br from-[#4f46e5] via-[#4338ca] to-[#312e81] p-10 shadow-[0_25px_60px_rgba(79,70,229,0.25)] text-white">
                    <div className="absolute top-0 right-0 w-96 h-96 bg-white/10 rounded-full blur-3xl -mr-20 -mt-20" />
                    <div className="absolute bottom-0 left-0 w-64 h-64 bg-cyan-400/10 rounded-full blur-3xl -ml-20 -mb-20" />
                    
                    <div className="relative z-10 flex flex-col md:flex-row md:items-center justify-between gap-8">
                        <div className="flex items-center gap-6">
                            <div className="w-20 h-20 rounded-[30px] bg-white/15 backdrop-blur-xl border border-white/10 flex items-center justify-center shadow-2xl">
                                <Activity className="w-10 h-10 text-white" />
                            </div>
                            <div>
                                <div className="flex items-center gap-3 mb-2">
                                    <button 
                                      onClick={() => navigate('/ecg-recus')}
                                      className="px-3 py-1 rounded-full bg-white/15 backdrop-blur-md border border-white/10 text-white text-[10px] font-black uppercase tracking-widest flex items-center gap-1.5 hover:bg-white/25 transition-colors"
                                    >
                                      <ArrowLeft className="w-3 h-3" /> Retour
                                    </button>
                                    <div className="px-3 py-1 rounded-full bg-emerald-400/20 text-emerald-100 text-[10px] font-black uppercase tracking-widest border border-emerald-300/10">
                                        ID: {analysis._id.slice(-6).toUpperCase()}
                                    </div>
                                </div>
                                <h1 className="text-4xl font-black tracking-tight">{analysis.ecg?.patient?.fullName || 'Examen ECG'}</h1>
                                <p className="text-indigo-100 mt-1 text-[15px] font-medium opacity-80">
                                    {analysis.ecg?.title} — {new Date(analysis.ecg?.createdAt || "").toLocaleDateString('fr-FR', { day: '2-digit', month: 'long', year: 'numeric' })}
                                </p>
                            </div>
                        </div>
                        <div className="flex items-center gap-4">
                            <ExportPDFButton analysisId={analysis._id} disabled={!isAnalyzed} />
                        </div>
                    </div>
                </div>

                <div className="grid grid-cols-1 lg:grid-cols-12 gap-8 items-start">
                    
                    {/* ── LEFT COLUMN ── */}
                    <div className="lg:col-span-8 space-y-8">
                        
                        {/* DIGITALIZED SIGNAL SLIDES */}
                        <PremiumCard className="p-8">
                            <div className="flex items-center justify-between mb-8">
                                <div className="flex items-center gap-3">
                                    <div className="w-12 h-12 rounded-2xl bg-indigo-50 flex items-center justify-center text-indigo-600 shadow-sm border border-indigo-100">
                                        <Zap className="w-6 h-6" />
                                    </div>
                                    <h2 className="text-2xl font-black text-slate-800 tracking-tight">Signal Digitalisé</h2>
                                </div>
                                {!isDigitized && (
                                    <Button onClick={handleDigitize} disabled={digitizing} className="h-12 px-6 rounded-2xl bg-indigo-600 hover:bg-indigo-700 shadow-lg shadow-indigo-600/20 gap-2">
                                        {digitizing ? <Loader2 className="w-4 h-4 animate-spin" /> : <Sparkles className="w-4 h-4" />}
                                        Lancer Digitalisation
                                    </Button>
                                )}
                            </div>

                            {isDigitized ? (
                                <div className="space-y-6">
                                    <div className="relative group rounded-[32px] bg-slate-50 border border-slate-100 p-2 overflow-hidden shadow-inner">
                                        <AnimatePresence mode="wait">
                                            <motion.img 
                                                key={activePlot}
                                                initial={{ opacity: 0, x: 20 }}
                                                animate={{ opacity: 1, x: 0 }}
                                                exit={{ opacity: 0, x: -20 }}
                                                src={plots[activePlot]?.url} 
                                                alt={plots[activePlot]?.label} 
                                                className="w-full h-auto rounded-[28px] bg-white shadow-sm"
                                            />
                                        </AnimatePresence>
                                        
                                        {plots.length > 1 && (
                                          <>
                                            <button 
                                              onClick={() => setActivePlot(prev => (prev > 0 ? prev - 1 : plots.length - 1))}
                                              className="absolute left-6 top-1/2 -translate-y-1/2 w-12 h-12 rounded-2xl bg-white/80 backdrop-blur-xl border border-white shadow-xl flex items-center justify-center text-indigo-600 opacity-0 group-hover:opacity-100 transition-all hover:scale-110 active:scale-95"
                                            >
                                              <ChevronLeft className="w-6 h-6" />
                                            </button>
                                            <button 
                                              onClick={() => setActivePlot(prev => (prev < plots.length - 1 ? prev + 1 : 0))}
                                              className="absolute right-6 top-1/2 -translate-y-1/2 w-12 h-12 rounded-2xl bg-white/80 backdrop-blur-xl border border-white shadow-xl flex items-center justify-center text-indigo-600 opacity-0 group-hover:opacity-100 transition-all hover:scale-110 active:scale-95"
                                            >
                                              <ChevronRight className="w-6 h-6" />
                                            </button>
                                          </>
                                        )}
                                    </div>
                                    
                                    <div className="flex items-center justify-center gap-3">
                                      {plots.map((p, i) => (
                                        <button 
                                          key={i}
                                          onClick={() => setActivePlot(i)}
                                          className={`px-4 py-2 rounded-xl text-[10px] font-black uppercase tracking-widest transition-all ${i === activePlot ? 'bg-indigo-600 text-white shadow-lg shadow-indigo-600/30 scale-105' : 'bg-slate-100 text-slate-400 hover:bg-slate-200'}`}
                                        >
                                          {p.label}
                                        </button>
                                      ))}
                                    </div>
                                </div>
                            ) : (
                                <div className="h-64 rounded-[32px] bg-slate-50 border-2 border-dashed border-slate-200 flex flex-col items-center justify-center gap-3 text-slate-400">
                                    <Microscope className="w-12 h-12 opacity-20" />
                                    <p className="text-sm font-bold uppercase tracking-widest opacity-40">Attente de digitalisation</p>
                                </div>
                            )}
                        </PremiumCard>

                        {/* ORIGINAL DOCUMENT */}
                        <PremiumCard className="p-8">
                            <div className="flex items-center gap-3 mb-8">
                                <div className="w-12 h-12 rounded-2xl bg-slate-50 flex items-center justify-center text-slate-600 shadow-sm border border-slate-100">
                                    <ImageIcon className="w-6 h-6" />
                                </div>
                                <h2 className="text-2xl font-black text-slate-800 tracking-tight">Document Original</h2>
                            </div>
                            <div className="p-2 bg-slate-900 rounded-[32px] overflow-hidden shadow-2xl">
                                <img src={imageUrl} alt="Papier original" className="w-full h-auto opacity-80 hover:opacity-100 transition-opacity rounded-[24px]" />
                            </div>
                        </PremiumCard>

                    </div>

                    {/* ── RIGHT COLUMN ── */}
                    <div className="lg:col-span-4 space-y-8 sticky top-8">
                        
                        {/* AI SUMMARY BOX */}
                        <PremiumCard className="p-0 border-none shadow-xl">
                            <div className={`p-8 ${isAnomalous ? 'bg-gradient-to-br from-rose-500 to-rose-600' : 'bg-gradient-to-br from-emerald-500 to-emerald-600'} text-white`}>
                                <div className="flex items-center justify-between mb-6">
                                    <div className="w-14 h-14 rounded-2xl bg-white/20 backdrop-blur-xl flex items-center justify-center shadow-lg border border-white/10">
                                        {isAnomalous ? <AlertCircle className="w-8 h-8" /> : <ShieldCheck className="w-8 h-8" />}
                                    </div>
                                    {!isAnalyzed && isDigitized && (
                                        <Button onClick={handleAnalyze} disabled={analyzing} className="bg-white/20 hover:bg-white/30 text-white border-0 backdrop-blur-md rounded-xl font-black text-[10px] tracking-widest uppercase h-10">
                                            {analyzing ? <Loader2 className="w-4 h-4 animate-spin" /> : "ANALYSER"}
                                        </Button>
                                    )}
                                </div>
                                <h3 className="text-3xl font-black tracking-tight mb-2 uppercase">{aiStatus}</h3>
                                <p className="text-white/80 text-sm font-medium leading-relaxed">
                                    {isAnalyzed 
                                        ? `Le modèle IA a classifié cet examen comme étant ${isAnomalous ? "ANORMAL" : "NORMAL"} avec un indice de confiance de ${(metrics?.ai_classification?.confidence * 100).toFixed(2)}%.` 
                                        : "L'analyse automatique par intelligence artificielle n'a pas encore été effectuée sur cet examen."}
                                </p>
                            </div>
                            
                            {isAnalyzed && (
                              <div className="p-8 bg-white space-y-6">
                                {/* AI ANOMALIES LIST */}
                                {isAnomalous && metrics?.ai_classification?.anomalies?.length > 0 && (
                                  <div className="space-y-3">
                                    <h4 className="text-[10px] font-black text-slate-400 uppercase tracking-widest">Anomalies Détectées</h4>
                                    <div className="flex flex-wrap gap-2">
                                      {metrics.ai_classification.anomalies.map((an: string) => (
                                        <span key={an} className="px-3 py-1.5 bg-rose-50 text-rose-600 text-xs font-black rounded-lg border border-rose-100 uppercase tracking-wider flex items-center gap-2">
                                          <div className="w-1.5 h-1.5 rounded-full bg-rose-500 animate-pulse" />
                                          {an}
                                        </span>

                                    </div>
                                  </div>
                                )}

                                {/* CATEGORIES PROBABILITIES */}
                                {metrics?.ai_classification?.n1?.positives && (
                                  <div className="space-y-4 pt-4 border-t border-slate-50">
                                    {Object.entries(metrics.ai_classification.n1.positives).map(([key, list]: [string, any]) => (
                                      list.length > 0 && (
                                        <div key={key} className="space-y-2">
                                          <h5 className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{CATEGORY_LABELS[key] || key}</h5>
                                          <div className="flex flex-wrap gap-2">
                                            {list.map((code: string) => (
                                              <Badge key={code} variant="secondary" className="bg-slate-50 text-slate-600 border-slate-100 rounded-lg px-2 py-1 font-bold text-[11px]">{code}</Badge>
                                            ))}
                                          </div>
                                        </div>
                                      )
                                    ))}
                                  </div>
                                )}
                              </div>
                            )}
                        </PremiumCard>

                        {/* METRICS DASHBOARD */}
                        <PremiumCard className="p-8">
                            <div className="flex items-center gap-3 mb-8">
                                <div className="w-12 h-12 rounded-2xl bg-indigo-50 flex items-center justify-center text-indigo-600 shadow-sm border border-indigo-100">
                                    <TrendingUp className="w-6 h-6" />
                                </div>
                                <h3 className="text-xl font-black text-slate-800 tracking-tight">Paramètres Vitaux</h3>
                            </div>
                            
                            <div className="grid grid-cols-2 gap-4 mb-4">
                                <MetricBox 
                                    label="Rythme" 
                                    value={metrics?.rhythm || "—"} 
                                    icon={Activity}
                                    status={metrics?.rhythm?.toLowerCase().includes('sinus') ? 'normal' : 'warning'}
                                />
                                <MetricBox 
                                    label="Fréquence" 
                                    value={metrics?.hr} 
                                    unit="BPM" 
                                    icon={Heart}
                                    status={metrics?.hr && (metrics.hr > 100 || metrics.hr < 50) ? 'warning' : 'normal'}

                            <div className="space-y-4 pt-4 border-t border-slate-50">
                                <div className="grid grid-cols-1 gap-4">

                                    <MetricBox label="Durée QRS" value={metrics?.qrs} unit="ms" />
                                    <MetricBox label="Intervalle QTc" value={metrics?.qtc} unit="ms" />

                                onClick={handleSaveNotes}

                                Enregistrer

                    </div>

            </div>
        </DashboardLayout>
    );
}
